import cv2
import os
import time
import pyautogui


def play_playlist(playlist_id):

    uri = f"spotify:playlist:{playlist_id}"

    print(f"Opening {uri}")

    os.startfile(uri)

    time.sleep(5)

    box = pyautogui.locateOnScreen(
        "resources/play_button.png",
        confidence=0.8
    )

    if box:
        pyautogui.click(
            box.left + 30,
            box.top + 30
        )
        print("Playing playlist")

    else:
        print("Play button not found")
